% ======================== 模板选项说明 ========================
% 本模板的主要行为通过 \documentclass[...] 里的选项控制。
%
% 1. 页面模式（旧功能）
%    single：电子版/单面模式，不额外插入装订用空白页。
%    print ：打印版模式，封面、声明等前置页之后会按双面打印习惯补空白页。
%
% 2. 封面格式（新增功能）
%    cover=2024  ：新版封面，默认值。包含标题上方的学位类型标识，
%                  并在下方信息表中显示“培养方式”。
%    cover=legacy：旧版封面，用于兼容旧版封皮扉页格式。
%                  旧版学术学位封面不显示“（学术学位）”，
%                  下方信息表也不显示“培养方式”。
%
% 3. 学位类型（新增功能）
%    degree=academic     ：学术学位，默认值。
%    degree=professional ：专业学位。新版封面会显示“（专业学位）”；
%                          旧版封面也会在标题上方显示专业学位标识，
%                          以便专业学位论文仍能明确区分。
%
%    如果学校后续要求使用特殊文字，也可以继续用旧兼容命令：
%      \DegreeType{（自定义学位类型）}
%    但一般情况下推荐优先使用 degree=academic/professional。
%
% 4. 原创性声明页（新增功能）
%    statement   ：生成原创性声明页，默认值。
%    nostatement ：不生成原创性声明页。适合只导出正文、盲审版，
%                  或者封面/声明页由学院系统另行生成的情况。
%
% 常用写法示例：
%   新版学术学位，含声明页：
%     \documentclass[single,print,cover=2024,degree=academic,statement]{sduthesis}
%   旧版学术学位封面，不含声明页：
%     \documentclass[single,print,cover=legacy,degree=academic,nostatement]{sduthesis}
%   新版专业学位，含声明页：
%     \documentclass[single,print,cover=2024,degree=professional,statement]{sduthesis}
% ==============================================================
\documentclass[single,print,cover=2024,degree=academic,statement]{sduthesis}
