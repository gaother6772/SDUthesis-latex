\begin{aligned}
\mathbb{E}[(X+Y)^2] &= \mathbb{E}[X^2] + 2\mathbb{E}[XY] + \mathbb{E}[Y^2] \\
&\leq \mathbb{E}[X^2] + 2\sqrt{\mathbb{E}[X^2]\mathbb{E}[Y^2]} + \mathbb{E}[Y^2] \\
&\leq 2\,\mathbb{E}[X^2] + 2\,\mathbb{E}[Y^2].
\end{aligned}
\end{equation}
