\begin{symboldescription}
	\begin{tabularx}{\textwidth}{X X}
		\(x\) & 输入变量或状态变量 \\
		\(y\) & 输出变量或目标变量 \\
		\(\hat{y}\) & 模型预测值 \\
		\(\theta\) & 模型参数 \\
		\(\mathcal{L}\) & 损失函数 \\
		\(\nabla\) & 梯度算子 \\
		\(\mathbb{E}[\cdot]\) & 数学期望 \\
		\(\mathbb{P}\) & 概率测度 \\
		\(\mathcal{N}(\mu, \Sigma)\) & 高斯分布 \\
		\(\sigma(\cdot)\) & 激活函数或标准差 \\
		\(\alpha\) & 学习率 \\
		\(\lambda\) & 正则化系数 \\
		\(\mathbb{R}^n\) & \(n\) 维欧氏空间 \\
		\(f_\theta\) & 参数为 \(\theta\) 的模型函数 \\
	\end{tabularx}
\end{symboldescription}
